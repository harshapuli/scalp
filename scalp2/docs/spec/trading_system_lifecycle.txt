1. Executive Summary
This project ships the Edge-Centric Trading System v2.2 from zero infrastructure to full-Kelly live trading on S5 Gamma Reversal.
1.1 Why three layers
The original Jira lifecycle (v1.1) covered only the S5 Strategy layer, assuming Scalp Brain and Swing Brain were already built. They aren't. This document fills that gap by including all three layers in dependency order.
1.2 Scalp Brain — full v2.2 §B4 implementation
The scalp brain is genuinely shared infrastructure across all five future strategies (S1-S5). Half-building it for S5 only would force re-work later. So we build the full 7-state classifier with all four score formulas (reversal, ignition, continuation, climax) for both directions (surge/tank).
1.3 Swing Brain — S5-only scope initially
The swing brain's 7-stage lifecycle (Inducement → Setup → Trigger → Entry → Manage → Exit → Journal) is conceptually shared, but each strategy's setup conditions are strategy-specific. We implement the lifecycle scaffolding once, plus the S5 setup/trigger/entry/manage logic. S1-S4 setup logic gets added incrementally as we ship each strategy.
1.4 S5 Strategy — production lifecycle
Phase 0 sub-edge validation through Phase 7 full-Kelly live deployment. Includes the ChatGPT-reviewed ML transition model with trade-outcome labeling, execution-aware features, EV-curve threshold selection, and per-direction posterior tracking.

def allow_s5_trade(features, model, threshold) -> Decision:
    if not features.is_s5_setup:
        return PASS('setup_failed')
    if features.reversal_score < features.required_reversal_score:
        return PASS('reversal_score_too_low')
    if features.gamma_distance_atr > 0.50:
        return PASS('not_near_gamma_strike')
    if features.event_blackout:
        return PASS('event_blackout')
    if features.gex_stale:
        return PASS('gex_stale')
    if features.bid_ask_spread_pct > 0.05:
        return PASS('option_spread_too_wide')
    if features.option_volume < features.min_option_volume:
        return PASS('option_liquidity_too_low')
    prob_success = model.predict_proba(features.ml_vector)[0][1]
    if prob_success < threshold:
        return PASS('ml_probability_below_threshold')
    if features.expected_value_net <= 0:
        return PASS('negative_ev')
    if not features.risk_manager_allows:
        return PASS('risk_denied')
    return TRADE('S5_GAMMA_REVERSAL', prob_success=prob_success)

S5 Setup Gate (v2.2 §14)
def is_s5_setup(f: Features, cfg: dict) -> tuple[bool, Optional[str]]:
    """Returns (eligible, reason_if_not)."""
    if f.distance_to_major_pos_gex_atr > cfg['s5']['setup']['near_gex_atr']:
        return False, 'not_near_gex'
    extended = (f.extension_from_vwap_atr >= cfg['s5']['setup']['extended_atr_min']
                or f.extension_from_prior_close_atr >= cfg['s5']['setup']['extended_atr_min'])
    if not extended:
        return False, 'not_extended'
    if f.near_htf_level_atr > cfg['s5']['setup']['at_liquidity_atr']:
        return False, 'not_at_liquidity'
    if f.earnings_blackout or f.fomc_blackout:
        return False, 'event_blackout'
    if f.major_gex_strike_dte < cfg['s5']['setup']['gex_strike_dte_min']:
        return False, 'gex_strike_too_close_to_expiry'
    if f.gex_snapshot_age_min > cfg['s5']['setup']['gex_snapshot_max_age_min']:
        return False, 'gex_stale'
    return True, None
S5 Trigger with Pierce Check (v2.2 §15)
def s5_trigger(pre_staged: Candidate, scalp_state: ScalpState,
               recent_bars: list[Bar], cfg: dict) -> tuple[bool, Optional[str]]:
    """Returns (should_trigger, reason_if_not)."""
    buf = cfg['s5']['trigger']['pierce_buffer_atr'] * pre_staged.atr
    # v2.2: invalidate if price already pierced the gamma strike
    if pre_staged.direction == 'short':
        pierced = max(b.high for b in recent_bars) > pre_staged.gamma_strike + buf
        if pierced:
            return False, 'gamma_pierced'
        rule_trigger = (scalp_state.name == 'SURGE_REVERSE'
                        and scalp_state.score >= cfg['s5']['trigger']['surge_reverse_score_min'])
    else:  # long
        pierced = min(b.low for b in recent_bars) < pre_staged.gamma_strike - buf
        if pierced:
            return False, 'gamma_pierced'
        rule_trigger = (scalp_state.name == 'TANK_REVERSE'
                        and scalp_state.score >= cfg['s5']['trigger']['tank_reverse_score_min'])
    if not rule_trigger:
        return False, 'scalp_state_not_reversing'
    return True, None
S5 Trade-Outcome Labeler (v2.2 §17 + ChatGPT review)
def label_s5_trade_outcome(
    row: CandidateRow,
    future_bars: list[Bar],
    direction: Literal['long_after_tank', 'short_after_surge'],
    decision_time_atr: float,
    decision_time_gex: GEXSnapshot,
    cost_model: CostModel,
) -> tuple[int, int]:
    """Returns (label_reversal_next_N, label_trade_success).
    CRITICAL look-ahead controls (v2.2):
    - decision_time_atr  : ATR computed using ONLY bars up to row.idx
    - decision_time_gex  : GEX snapshot dated <= row.timestamp
    - future_bars        : strictly bars[row.idx+1 : row.idx+1+N]
    label_reversal_next_N is kept for sub-edge attribution (Phase 0).
    label_trade_success is the production label (per ChatGPT review):
    1 only if simulated production trade reaches target, survives stops,
    timeouts, IV crush, AND clears modeled costs.
    """
    atr = decision_time_atr
    gex = decision_time_gex.major_strike_at(row.timestamp)
    buf = 0.25 * atr
    # ---- label_reversal_next_N (v2.2 §17 reversal-state label) ----
    if direction == 'short_after_surge':
        hit_target = min(b.low for b in future_bars) <= row.close - 1.0 * atr
        invalidated = max(b.high for b in future_bars) >= gex + buf
        label_reversal = 1 if hit_target and not invalidated else 0
    else:  # long_after_tank
        hit_target = max(b.high for b in future_bars) >= row.close + 1.0 * atr
        invalidated = min(b.low for b in future_bars) <= gex - buf
        label_reversal = 1 if hit_target and not invalidated else 0
    # ---- label_trade_success (production, ChatGPT review) ----
    sim_entry = row.next_bar_open + cost_model.entry_slippage(row)
    sim_target_price = sim_entry + (1.0 * atr if direction.startswith('long') else -1.0 * atr)
    sim_stop_price   = sim_entry - (1.0 * atr if direction.startswith('long') else -1.0 * atr)
    target_hit_first = simulate_first_touch(future_bars, sim_target_price, sim_stop_price, direction)
    if not target_hit_first:
        return label_reversal, 0
    # IV crush check (vertical-spread aware)
    iv_at_target = future_bars[index_of_target_hit].atm_iv
    iv_crush_pct = (row.iv_at_decision - iv_at_target) / row.iv_at_decision
    if iv_crush_pct > cost_model.iv_crush_kill_pct:
        return label_reversal, 0
    sim_pnl = (sim_target_price - sim_entry) * row.contracts * 100  # dollars
    sim_pnl_net = sim_pnl - cost_model.total_cost(row)
    return label_reversal, (1 if sim_pnl_net > 0 else 0)
EV-Curve Threshold Picker (v2.2 §19)
def pick_threshold(probabilities: np.ndarray, labels: np.ndarray,
                   costs_per_trade: float, payoff_per_winner: float,
                   cfg: dict) -> tuple[float, float]:
    """Derive threshold from validation EV curve, not prior. Per v2.2 §19."""
    best_thr, best_ev = cfg['s5']['ml']['threshold_low'], -float('inf')
    min_trades = cfg['s5']['ml']['min_validation_trades_per_threshold']
    for thr in np.arange(cfg['s5']['ml']['threshold_low'],
                         cfg['s5']['ml']['threshold_high'], 0.01):
        traded = probabilities >= thr
        if traded.sum() < min_trades:
            continue
        wins   = (traded & (labels == 1)).sum()
        losses = (traded & (labels == 0)).sum()
        ev = wins * payoff_per_winner - losses - traded.sum() * costs_per_trade
        ev_per_trade = ev / traded.sum()
        if ev_per_trade > best_ev:
            best_thr, best_ev = thr, ev_per_trade
    if best_thr <= cfg['s5']['ml']['threshold_extreme_warn_low'] or \
       best_thr >= cfg['s5']['ml']['threshold_extreme_warn_high']:
        warn(f'threshold {best_thr} at extreme; investigate calibration')
    return best_thr, best_ev
S5 Order Builder (v2.2 §21)
def build_s5_order(candidate: Candidate, chain: OptionChain,
                   cfg: dict) -> Decision:
    expiry = choose_expiry(chain,
                           min_dte=cfg['s5']['execution']['dte_min'],
                           max_dte=cfg['s5']['execution']['dte_max'])
    side = 'put' if candidate.direction == 'short' else 'call'
    long_leg = choose_delta_leg(chain, expiry,
                                delta_min=cfg['s5']['execution']['delta_min'],
                                delta_max=cfg['s5']['execution']['delta_max'],
                                side=side)
    short_leg = choose_short_leg(long_leg, target_width=candidate.spread_width)
    debit = mid_price(long_leg) - mid_price(short_leg)
    width = spread_width(long_leg, short_leg)
    if debit / width > cfg['s5']['execution']['debit_to_width_max']:
        return PASS('debit_too_expensive', strategy='S5')
    if option_spread_pct(long_leg, short_leg) > cfg['s5']['execution']['option_spread_pct_max']:
        return PASS('spread_too_wide', strategy='S5')
    return TRADE('S5',
        instrument=VerticalOrder(long_leg=long_leg, short_leg=short_leg,
            limit_price=debit * cfg['s5']['execution']['limit_buffer'],
            ...))
Per-Direction Posterior Update (S5-61)
@dataclass
class DirectionalPosterior:
    direction: Literal['long_after_tank', 'short_after_surge']
    alpha: float
    beta: float
    @property
    def hit_rate(self) -> float:
        return self.alpha / (self.alpha + self.beta)
    def update(self, won: bool):
        if won: self.alpha += 1
        else:   self.beta += 1
    def divergence_from(self, other: 'DirectionalPosterior') -> float:
        return abs(self.hit_rate - other.hit_rate)
def check_drift(post: DirectionalPosterior, prior_hit_rate: float, cfg: dict):
    drop_pp = (prior_hit_rate - post.hit_rate) * 100
    if drop_pp > cfg['s5']['posterior']['drift_alert_pp']:
        alert(f'{post.direction} posterior dropped {drop_pp:.1f}pp')
Scalp Brain Reversal Score (v2.2 §B4.3 — starter weights)
def reversal_score(f: Features, direction: Literal['long', 'short'], cfg: dict) -> float:
    """Returns score in [0, 1]. Calibrate weights via SCALP-10.3.
    Long direction = TANK_REVERSE (bottom reversal forming).
    Short direction = SURGE_REVERSE (top reversal forming)."""
    w = cfg['scalp']['reversal_score']
    if direction == 'short':  # SURGE_REVERSE: looking for top reversal
        failed_ext = clamp01(-f.failed_extension_atr / 1.0)        # negative means retraced
        wick = clamp01(f.upper_wick_pct / 0.5)
        agg_flip = clamp01((f.aggressor_prior - f.aggressor_recent) / 1.0)  # flip down
        vol_div = clamp01(1 - f.volume_divergence_ratio)           # current < prior
        flow_flip = 1.0 if f.flow_flip and f.net_signed_premium_5m < 0 else 0.0
    else:  # 'long' = TANK_REVERSE: looking for bottom reversal
        failed_ext = clamp01(f.failed_extension_atr / 1.0)
        wick = clamp01(f.lower_wick_pct / 0.5)
        agg_flip = clamp01((f.aggressor_recent - f.aggressor_prior) / 1.0)  # flip up
        vol_div = clamp01(1 - f.volume_divergence_ratio)
        flow_flip = 1.0 if f.flow_flip and f.net_signed_premium_5m > 0 else 0.0
    score = (w['weight_failed_extension'] * failed_ext
             + w['weight_wick'] * wick
             + w['weight_aggressor_flip'] * agg_flip
             + w['weight_volume_divergence'] * vol_div
             + w['weight_flow_flip'] * flow_flip)
    return clamp01(score)
